agent.ts TYPESCRIPT 
import { defineAgent, route } from "@apex/sdk";

export default defineAgent({
  id: "atlas-research",
  runtime: "agent@2",
  tools: [web.search, files.read, memory],

  // 按成本优先路由，失败自动切换
  model: route({
    strategy: "cost",
    fallback: ["gpt-5", "claude-4.5"],
  }),

  async run(ctx, input) {
    const plan = await ctx.plan(input.task);
    const sources = await ctx.gather(plan, {
      citations: true,
    });
    return ctx.report(sources);
  },
});
